
#include <stdio.h>

#define MAX 20

void display(int n, int pid[], int bt[],
             int wt[], int tat[]) {
    int i;
    float totalWT = 0, totalTAT = 0;

    printf("\nPID\tBurst\tWaiting\tTurnaround\n");

    for (i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\n",
               pid[i], bt[i], wt[i], tat[i]);

        totalWT += wt[i];
        totalTAT += tat[i];
    }

    printf("\nAverage Waiting Time: %.2f\n",
           totalWT / n);
    printf("Average Turnaround Time: %.2f\n",
           totalTAT / n);
}

void fcfs(int n, int pid[], int bt[]) {
    int wt[MAX], tat[MAX];
    int i;

    wt[0] = 0;

    for (i = 1; i < n; i++)
        wt[i] = wt[i - 1] + bt[i - 1];

    for (i = 0; i < n; i++)
        tat[i] = wt[i] + bt[i];

    printf("\n--- FCFS Scheduling ---\n");
    printf("Execution order: ");

    for (i = 0; i < n; i++)
        printf("P%d ", pid[i]);

    printf("\n");
    display(n, pid, bt, wt, tat);
}

void sjf(int n, int pid[], int bt[]) {
    int p[MAX], b[MAX];
    int wt[MAX], tat[MAX];
    int i, j, temp;

    for (i = 0; i < n; i++) {
        p[i] = pid[i];
        b[i] = bt[i];
    }

    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {
            if (b[j] > b[j + 1]) {
                temp = b[j];
                b[j] = b[j + 1];
                b[j + 1] = temp;

                temp = p[j];
                p[j] = p[j + 1];
                p[j + 1] = temp;
            }
        }
    }

    wt[0] = 0;

    for (i = 1; i < n; i++)
        wt[i] = wt[i - 1] + b[i - 1];

    for (i = 0; i < n; i++)
        tat[i] = wt[i] + b[i];

    printf("\n--- SJF Scheduling ---\n");
    printf("Execution order: ");

    for (i = 0; i < n; i++)
        printf("P%d ", p[i]);

    printf("\n");
    display(n, p, b, wt, tat);
}

void roundRobin(int n, int pid[], int bt[], int quantum) {
    int rem[MAX], wt[MAX], tat[MAX], ct[MAX];
    int i, time = 0, done;

    for (i = 0; i < n; i++)
        rem[i] = bt[i];

    printf("\n--- Round Robin Scheduling ---\n");
    printf("Execution order: ");

    do {
        done = 1;

        for (i = 0; i < n; i++) {
            if (rem[i] > 0) {
                done = 0;
                printf("P%d ", pid[i]);

                if (rem[i] > quantum) {
                    time += quantum;
                    rem[i] -= quantum;
                } else {
                    time += rem[i];
                    rem[i] = 0;
                    ct[i] = time;
                }
            }
        }
    } while (!done);

    printf("\n");

    for (i = 0; i < n; i++) {
        tat[i] = ct[i];
        wt[i] = tat[i] - bt[i];
    }

    display(n, pid, bt, wt, tat);
}

int main() {
    int pid[MAX], bt[MAX];
    int n, i, choice, quantum;

    printf("=== CPU SCHEDULING SIMULATOR ===\n");
    printf("Enter number of processes (1-%d): ", MAX);
    scanf("%d", &n);

    if (n < 1 || n > MAX) {
        printf("Invalid number of processes!\n");
        return 1;
    }

    for (i = 0; i < n; i++) {
        pid[i] = i + 1;

        printf("Enter burst time for P%d: ", pid[i]);
        scanf("%d", &bt[i]);

        if (bt[i] <= 0) {
            printf("Burst time must be positive!\n");
            return 1;
        }
    }

    do {
        printf("\n===== MENU =====\n");
        printf("1. FCFS Scheduling\n");
        printf("2. SJF Scheduling\n");
        printf("3. Round Robin Scheduling\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                fcfs(n, pid, bt);
                break;

            case 2:
                sjf(n, pid, bt);
                break;

            case 3:
                printf("Enter time quantum: ");
                scanf("%d", &quantum);

                if (quantum <= 0) {
                    printf("Quantum must be positive!\n");
                } else {
                    roundRobin(n, pid, bt, quantum);
                }
                break;

            case 4:
                printf("Exiting simulator.\n");
                break;

            default:
                printf("Invalid choice!\n");
        }
    } while (choice != 4);

    return 0;
}
